# 1. Январская валидация: клиенты, отложенные в OOS до обучения.
features_check = list(best_model.feature_name_)

train_client_ids = set(train['client_did'])
oos_client_ids_check = set(oos['client_did'])

assert train_client_ids.isdisjoint(oos_client_ids_check)
assert test['client_did'].notna().all()

# Проверяем, что test действительно содержит только январь.
assert (
    pd.to_datetime(test['report_date']).dt.to_period('M')
    == pd.Period('2026-01', freq='M')
).all()

jan_check = test.copy()
jan_check['client_group_check'] = np.select(
    [
        jan_check['client_did'].isin(train_client_ids),
        jan_check['client_did'].isin(oos_client_ids_check),
    ],
    ['seen_in_train', 'held_out_oos'],
    default='new_client'
)

jan_valid = jan_check.loc[
    jan_check['client_group_check'].eq('held_out_oos')
].copy()

assert not jan_valid.empty
assert jan_valid[TARGET].nunique() == 2
assert set(jan_valid['client_did']).isdisjoint(train_client_ids)

print(
    f'Валидация: {len(jan_valid):,} контрактных строк, '
    f'{jan_valid["client_did"].nunique():,} клиентов, '
    f'доля таргета {jan_valid[TARGET].mean():.2%}'
)


# 2. Все параметры прежние; меняем только eval_set.
# Возвращаем верхний предел вместо найденных ранее 2985 деревьев.
params_check = best_model.get_params().copy()
params_check.update({
    'n_estimators': 3000,
    'objective': 'binary',
    'metric': 'auc',
})

model_group_valid = lgb.LGBMClassifier(**params_check)

model_group_valid.fit(
    train[features_check],
    train[TARGET],
    eval_set=[(
        jan_valid[features_check],
        jan_valid[TARGET]
    )],
    eval_metric='auc',
    callbacks=[
        lgb.early_stopping(
            stopping_rounds=100,
            first_metric_only=True,
            verbose=False
        )
    ]
)

group_best_iteration = int(model_group_valid.best_iteration_)
print(f'Выбрано деревьев: {group_best_iteration}')

# Проверяем, хватило ли верхнего предела для полного ожидания.
evaluated_rounds = len(
    model_group_valid.evals_result_['valid_0']['auc']
)
if (
    evaluated_rounds >= params_check['n_estimators']
    and evaluated_rounds - group_best_iteration < 100
):
    print(
        'Достигнут лимит деревьев до завершения '
        '100 итераций ожидания после лучшего результата.'
    )


# 3. Сравнение трёх вариантов на одинаковых строках.
old_full_iteration = best_model.booster_.current_iteration()
assert old_full_iteration >= 800

variants = [
    ('current_800', best_model, 800),
    ('current_full', best_model, old_full_iteration),
    ('group_early_stopping', model_group_valid, group_best_iteration),
]

datasets = {
    'train': train,
    'oos_sep_dec': oos,
    'jan_all': jan_check,
}

for group_name in ['seen_in_train', 'held_out_oos', 'new_client']:
    datasets[f'jan_{group_name}'] = jan_check.loc[
        jan_check['client_group_check'].eq(group_name)
    ]

rows = []

for model_name, model, n_trees in variants:
    for segment, data in datasets.items():
        auc = np.nan

        if not data.empty and data[TARGET].nunique() == 2:
            pred = model.predict_proba(
                data[features_check],
                num_iteration=n_trees
            )[:, 1]
            auc = roc_auc_score(data[TARGET], pred)

        rows.append({
            'model': model_name,
            'n_trees': n_trees,
            'segment': segment,
            'n_samples': len(data),
            'roc_auc': auc,
            'gini': 2 * auc - 1,
        })

comparison_group_valid = pd.DataFrame(rows)

auc_comparison = (
    comparison_group_valid
    .pivot(index='segment', columns='model', values='roc_auc')
    .reindex(index=list(datasets))
    .reindex(columns=[name for name, _, _ in variants])
)

auc_comparison['delta_vs_800'] = (
    auc_comparison['group_early_stopping']
    - auc_comparison['current_800']
)

auc_comparison['delta_vs_full'] = (
    auc_comparison['group_early_stopping']
    - auc_comparison['current_full']
)

display(auc_comparison.round(6))
