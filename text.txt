import numpy as np
import pandas as pd
from sklearn.metrics import roc_auc_score

features_diag = list(best_model.feature_name_)

# Для сравнения каждому контракту должна соответствовать одна строка за месяц.
for month_diag, frame_diag in [('2026-01', test), ('2026-02', oot)]:
    assert frame_diag['contract_uid'].notna().all(), 'Есть пустые contract_uid'
    assert frame_diag['contract_uid'].is_unique, (
        f'{month_diag}: несколько строк на один контракт'
    )
    assert (
        frame_diag['report_date'].dt.strftime('%Y-%m') == month_diag
    ).all(), f'В выборке есть даты вне {month_diag}'

common_contracts = pd.Index(test['contract_uid']).intersection(
    pd.Index(oot['contract_uid'])
)

cohort_rows = []

for month_diag, frame_diag, only_group in [
    ('2026-01', test, 'only_january'),
    ('2026-02', oot, 'only_february'),
]:
    # Один расчёт прогнозов на весь месяц.
    pred_diag = best_model.predict_proba(frame_diag[features_diag])[:, 1]

    common_mask = frame_diag['contract_uid'].isin(common_contracts).to_numpy()

    for group_diag, mask_diag in [
        ('all', np.ones(len(frame_diag), dtype=bool)),
        ('same_contracts', common_mask),
        (only_group, ~common_mask),
    ]:
        y_diag = frame_diag.loc[mask_diag, TARGET]

        cohort_rows.append({
            'month': month_diag,
            'group': group_diag,
            'n_contracts': int(mask_diag.sum()),
            'share_of_month': float(mask_diag.mean()),
            'target_share': y_diag.mean(),
            'auc': (
                roc_auc_score(y_diag, pred_diag[mask_diag])
                if y_diag.nunique() == 2 else np.nan
            ),
        })

cohort_metrics = pd.DataFrame(cohort_rows)

print('Качество и состав групп:')
display(cohort_metrics.round(6))

# Сравнение месяцев на всей базе и на одинаковом наборе контрактов.
cohort_comparison = (
    cohort_metrics[
        cohort_metrics['group'].isin(['all', 'same_contracts'])
    ]
    .pivot(index='group', columns='month', values='auc')
    .reindex(['all', 'same_contracts'])
)

cohort_comparison['delta_feb_minus_jan'] = (
    cohort_comparison['2026-02'] - cohort_comparison['2026-01']
)

print('Изменение AUC: февраль минус январь:')
display(cohort_comparison.round(6))

del pred_diag
