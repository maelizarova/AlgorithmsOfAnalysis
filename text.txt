import pandas as pd
from sklearn.metrics import roc_auc_score

features_check = list(best_model.feature_name_)

for data_check in (test, oot):
    assert data_check['contract_uid'].notna().all(), 'Есть пустые contract_uid'
    assert data_check['contract_uid'].is_unique, 'Есть повторы contract_uid внутри месяца'

# Январские скоры исходной модели.
jan_score_check = test[['contract_uid']].copy()
jan_score_check['score_jan'] = best_model.predict_proba(
    test[features_check]
)[:, 1]

# Февральские скоры и февральский таргет.
feb_score_check = oot[['contract_uid', TARGET]].rename(
    columns={TARGET: 'target_feb'}
).copy()

feb_score_check['score_feb'] = best_model.predict_proba(
    oot[features_check]
)[:, 1]

# Оставляем только контракты, присутствующие в обоих месяцах.
paired_score_check = jan_score_check.merge(
    feb_score_check,
    on='contract_uid',
    how='inner',
    validate='one_to_one',
)

assert len(paired_score_check) > 0, 'Нет общих контрактов'
assert paired_score_check['target_feb'].notna().all(), 'Есть пропуски таргета'
assert paired_score_check['target_feb'].nunique() == 2, 'Для AUC нужны оба класса'

# В обоих расчётах — одни и те же контракты и один февральский таргет.
auc_jan_score = roc_auc_score(
    paired_score_check['target_feb'],
    paired_score_check['score_jan'],
)

auc_feb_score = roc_auc_score(
    paired_score_check['target_feb'],
    paired_score_check['score_feb'],
)

print(f'Общих контрактов: {len(paired_score_check):,}')

display(pd.DataFrame({
    'признаки': ['январские', 'февральские'],
    'таргет': ['февральский', 'февральский'],
    'auc': [auc_jan_score, auc_feb_score],
}).round(6))

print(
    'Изменение AUC при обновлении признаков: '
    f'{auc_feb_score - auc_jan_score:+.6f}'
)
