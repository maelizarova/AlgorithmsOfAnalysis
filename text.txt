# Одна новая ячейка. Нужны sample, train, oos, test, oot, best_model, TARGET.
# sample — исходный фрейм с open_dt и max_clip_dt; при необходимости замените
# только RAW_LIFECYCLE. Исходные фреймы и модель не меняются, обучения нет.
import numpy as np
import pandas as pd
from sklearn.metrics import roc_auc_score
from IPython.display import display

RAW_LIFECYCLE = sample
MODEL_LIFECYCLE = best_model  # исходная модель на 40 признаках
MONTHS_LIFECYCLE = pd.period_range("2025-09", "2026-02", freq="M").astype(str).tolist()
BATCH_LIFECYCLE = 50_000


def run_lifecycle_check(raw, parts, model, target):
    keys = ["contract_uid", "report_date"]
    features = list(model.booster_.feature_name())
    meta = pd.concat([
        frame[keys + [target]].rename(columns={target: "y"}).assign(source=name)
        for name, frame in parts
    ], ignore_index=True)
    meta["report_date"] = pd.to_datetime(meta["report_date"], errors="raise")
    meta["month"] = meta.report_date.dt.strftime("%Y-%m")
    assert meta[keys].notna().all().all(), "Пропуски в ключах модельной базы"
    assert meta.y.isin([0, 1]).all(), "Нужен бинарный таргет"
    assert set(meta.month) == set(MONTHS_LIFECYCLE), "Ожидались все месяцы сентября–февраля"
    assert not meta.duplicated(["contract_uid", "month"]).any(), "Повтор контракта внутри месяца"
    assert meta.loc[meta.source.isin(["train", "oos"]), "month"].lt("2026-01").all()
    assert meta.loc[meta.source.eq("jan"), "month"].eq("2026-01").all()
    assert meta.loc[meta.source.eq("feb"), "month"].eq("2026-02").all()

    # Только четыре столбца исходной базы, без копирования всех признаков.
    dates = raw[keys + ["open_dt", "max_clip_dt"]].copy()
    dates["report_date"] = pd.to_datetime(dates["report_date"], errors="raise")
    dates = dates.loc[dates.report_date.ge("2025-09-01") & dates.report_date.lt("2026-03-01")]
    assert dates[keys].notna().all().all(), "Пропуски в ключах исходной базы"
    assert not dates.duplicated(keys).any(), "Повтор контракт × report_date в исходной базе"
    meta = meta.merge(dates, on=keys, how="left", validate="one_to_one", indicator=True, sort=False)
    assert meta._merge.eq("both").all(), "Не все контракт × report_date найдены в RAW_LIFECYCLE"
    meta = meta.drop(columns="_merge")

    # Сравниваем календарные даты. Будущие/неразобранные даты не заменяем нулём.
    report_day = meta.report_date.dt.normalize()
    for prefix, col in [("age", "open_dt"), ("clip", "max_clip_dt")]:
        original = meta[col]
        parsed = pd.to_datetime(original, errors="coerce").dt.normalize()
        meta[prefix + "_missing"] = original.isna()
        meta[prefix + "_invalid"] = original.notna() & parsed.isna()
        meta[prefix + "_future"] = parsed.gt(report_day)
        meta[col] = parsed
    meta["clip_before_open"] = meta.max_clip_dt.lt(meta.open_dt)
    for prefix, col, edges, labels in [
        ("age", "open_dt", [0, 90, 180, 365, 730, np.inf],
         ["0–89", "90–179", "180–364", "365–729", "730+"]),
        ("clip", "max_clip_dt", [0, 30, 90, 180, 365, np.inf],
         ["0–29", "30–89", "90–179", "180–364", "365+"]),
    ]:
        days = (report_day - meta[col]).dt.days.astype(float)
        bad = meta[prefix + "_future"] | meta[prefix + "_invalid"] | meta[prefix + "_missing"]
        if prefix == "clip":
            bad |= meta.clip_before_open
        meta[prefix + "_days"] = days.mask(bad)
        bucket = pd.cut(meta[prefix + "_days"], edges, labels=labels, right=False).astype("string")
        bucket.loc[meta[prefix + "_missing"]] = "MISSING_DATE"
        bucket.loc[meta[prefix + "_invalid"]] = "INVALID_DATE"
        bucket.loc[meta[prefix + "_future"]] = "FUTURE_DATE"
        if prefix == "clip":
            bucket.loc[meta.clip_before_open] = "CLIP_BEFORE_OPEN"
        meta[prefix + "_bucket"] = bucket

    quality_flags = ["age_missing", "age_invalid", "age_future", "clip_missing",
                     "clip_invalid", "clip_future", "clip_before_open"]
    date_quality = meta.groupby("month")[quality_flags].mean().mul(100).add_suffix("_pct")
    date_quality.insert(0, "n_rows", meta.groupby("month").size())
    print("1. Качество дат; все доли в процентах, по всем строкам месяца:")
    display(date_quality.round(3))
    print("MISSING_DATE означает неизвестную дату, а не доказанное отсутствие клипов.")
    print("FUTURE_DATE: дата позже report_date; время после такого события не рассчитывается.")

    # Предсказания только для OOS / января / февраля, небольшими порциями.
    # Положение строк восстанавливаем по ключам, не полагаемся на порядок merge.
    meta["score"] = np.nan
    lookup = pd.MultiIndex.from_frame(meta[keys])
    for name, frame in parts:
        if name == "train":
            continue
        frame_keys = frame[keys].copy()
        frame_keys["report_date"] = pd.to_datetime(frame_keys.report_date)
        positions = lookup.get_indexer(pd.MultiIndex.from_frame(frame_keys))
        assert (positions >= 0).all()
        print(f"Рассчитываем прогнозы: {name}, {len(frame):,} строк", flush=True)
        for start in range(0, len(frame), BATCH_LIFECYCLE):
            end = min(start + BATCH_LIFECYCLE, len(frame))
            scores = model.predict_proba(frame.iloc[start:end][features])[:, 1]
            assert np.isfinite(scores).all(), "Неконечные прогнозы"
            meta.loc[positions[start:end], "score"] = scores

    def auc(frame):
        return roc_auc_score(frame.y, frame.score) if frame.y.nunique() == 2 else np.nan

    # Качество по месяцу и возрастному диапазону, без train в расчёте AUC.
    monthly_rows = []
    for month, whole in meta.groupby("month", sort=True):
        for dimension in ["all", "age", "clip"]:
            groups = [("all", whole)] if dimension == "all" else whole.groupby(
                dimension + "_bucket", observed=True, sort=False, dropna=False
            )
            for bucket, frame in groups:
                evaluation = frame.loc[frame.score.notna()]
                monthly_rows.append({
                    "month": month, "dimension": dimension, "bucket": bucket,
                    "n_all": len(frame), "target_all_pct": 100*frame.y.mean(),
                    "n_eval": len(evaluation), "n_pos_eval": int(evaluation.y.eq(1).sum()),
                    "n_neg_eval": int(evaluation.y.eq(0).sum()),
                    "target_eval_pct": 100*evaluation.y.mean(), "auc_eval": auc(evaluation),
                })
    monthly_bins = pd.DataFrame(monthly_rows)
    print("2. AUC: сентябрь–декабрь только OOS, январь — validation, февраль — OOT:")
    display(monthly_bins.loc[monthly_bins.dimension.eq("all"), [
        "month", "n_all", "target_all_pct", "n_eval", "auc_eval"
    ]].round(4))

    # Переходы: все общие контракты каждой соседней пары месяцев.
    # Диапазон возраста/давности клипа берём на ПЕРВУЮ дату пары.
    pair_info, profiles, rates, pair_auc_rows = [], [], [], []
    order = ["0 -> 0", "0 -> 1", "1 -> 0", "1 -> 1"]
    for before, after in zip(MONTHS_LIFECYCLE[:-1], MONTHS_LIFECYCLE[1:]):
        cols = ["contract_uid", "y", "score", "age_days", "clip_days", "age_bucket", "clip_bucket"]
        left = meta.loc[meta.month.eq(before), cols].set_index("contract_uid")
        right = meta.loc[meta.month.eq(after), ["contract_uid", "y", "score"]].set_index("contract_uid")
        pair = left.join(right, how="inner", lsuffix="_before", rsuffix="_after")
        name = f"{before} -> {after}"
        if pair.empty:
            print(f"Нет общих контрактов: {name}")
            continue
        pair["transition"] = pair.y_before.astype(int).astype(str) + " -> " + pair.y_after.astype(int).astype(str)
        evaluation = pair.loc[pair.score_after.notna()].copy()
        # Метка предыдущего месяца нужна для группировки, а не для обучения.
        # Для AUC достаточно, чтобы ПОСЛЕДУЮЩАЯ строка была вне train.
        for transition in order:
            group = pair.loc[pair.transition.eq(transition)]
            evaluated = evaluation.loc[evaluation.transition.eq(transition)]
            denominator = pair.y_before.eq(int(transition[0])).sum()
            profiles.append({
                "pair": name, "transition": transition, "n_all": len(group),
                "within_before_class_pct": 100*len(group)/denominator if denominator else np.nan,
                "age_before_median_days": group.age_days.median(),
                "clip_before_median_days": group.clip_days.median(),
                "age_unknown_or_invalid_pct": 100*group.age_days.isna().mean(),
                "clip_unknown_or_invalid_pct": 100*group.clip_days.isna().mean(),
                "n_eval": len(evaluated), "score_after_mean": evaluated.score_after.mean(),
            })
        for dimension in ["age", "clip"]:
            for bucket, group in pair.groupby(dimension + "_bucket", observed=True, sort=False, dropna=False):
                counts = group.transition.value_counts()
                n00, n01, n10, n11 = [int(counts.get(t, 0)) for t in order]
                rates.append({
                    "pair": name, "dimension": dimension, "bucket_before": bucket,
                    "n_common": len(group), "n_0_before": n00+n01, "n_1_before": n10+n11,
                    "n_0_to_1": n01, "n_1_to_0": n10,
                    "rate_0_to_1_pct": 100*n01/(n00+n01) if n00+n01 else np.nan,
                    "rate_1_to_0_pct": 100*n10/(n10+n11) if n10+n11 else np.nan,
                })

        positive_n = int(evaluation.y_after.eq(1).sum())
        negative_n = int(evaluation.y_after.eq(0).sum())
        total_pairs = positive_n * negative_n
        weighted_auc = 0.0
        for positive_group in ["0 -> 1", "1 -> 1"]:
            for negative_group in ["0 -> 0", "1 -> 0"]:
                selected = evaluation.loc[evaluation.transition.isin([positive_group, negative_group])]
                np_ = int(selected.y_after.eq(1).sum())
                nn_ = int(selected.y_after.eq(0).sum())
                cell_auc = roc_auc_score(selected.y_after, selected.score_after) if np_ and nn_ else np.nan
                weight = np_*nn_/total_pairs if total_pairs else np.nan
                if np_ and nn_:
                    weighted_auc += weight * cell_auc
                pair_auc_rows.append({
                    "pair": name, "positive_group": positive_group, "negative_group": negative_group,
                    "n_positive": np_, "n_negative": nn_, "auc": cell_auc,
                    "share_pos_neg_pairs_pct": 100*weight,
                })
        full_auc = roc_auc_score(evaluation.y_after, evaluation.score_after) if total_pairs else np.nan
        if total_pairs:
            assert np.isclose(weighted_auc, full_auc), "Не сошлось разложение AUC на четыре сравнения"
        pair_info.append({
            "pair": name, "n_common_all": len(pair), "n_common_eval": len(evaluation),
            "n_pos_eval": positive_n, "n_neg_eval": negative_n,
            "auc_after_common_eval": full_auc, "auc_weighted_four_groups": weighted_auc if total_pairs else np.nan,
        })

    transition_profile = pd.DataFrame(profiles)
    transition_rates = pd.DataFrame(rates)
    auc_pairs = pd.DataFrame(pair_auc_rows)
    auc_summary = pd.DataFrame(pair_info)
    if not auc_summary.empty:
        print("3. Четыре сравнения AUC; только отложенные строки ПОСЛЕДУЮЩЕГО месяца:")
        display(auc_pairs.pivot(index=["positive_group", "negative_group"], columns="pair", values="auc").round(4))
        print("Численности и проверка разложения; пересечение контрактов своё для каждой пары:")
        display(auc_summary.round(6))
        latest = auc_summary.pair.iloc[-1]
        print(f"4. Профили всех переходов: {latest}; возраст и клип на ПРЕДЫДУЩУЮ дату:")
        display(transition_profile.loc[transition_profile.pair.eq(latest)].round(3))
        print("Размеры четырёх сравнений AUC последней пары:")
        display(auc_pairs.loc[auc_pairs.pair.eq(latest)].round(4))
        for dimension, title in [("age", "Дней с открытия"), ("clip", "Дней после последнего клипа")]:
            print(f"5. {title}: частота 1 -> 0 СРЕДИ ИСХОДНЫХ ЕДИНИЦ, %:")
            subset = transition_rates.loc[transition_rates.dimension.eq(dimension)]
            display(subset.pivot(index="bucket_before", columns="pair", values="rate_1_to_0_pct").round(3))
            print("Частота 0 -> 1 СРЕДИ ИСХОДНЫХ НУЛЕЙ, %:")
            display(subset.pivot(index="bucket_before", columns="pair", values="rate_0_to_1_pct").round(3))
            print(f"Знаменатели и оба перехода: {latest}")
            display(subset.loc[subset.pair.eq(latest)].round(3))
            print(f"AUC внутри диапазонов «{title}»; каждый месяц отдельно:")
            subset = monthly_bins.loc[monthly_bins.dimension.eq(dimension)]
            display(subset.pivot(index="bucket", columns="month", values="auc_eval").round(4))

    print("Полные таблицы всех месяцев сохраняются в lifecycle_*. Расчётов на train для AUC нет.")
    print("Сравнения описательные: состав и размеры возрастных групп могут меняться по месяцам.")
    return date_quality, monthly_bins, transition_profile, transition_rates, auc_pairs, auc_summary


(lifecycle_date_quality, lifecycle_monthly_bins, lifecycle_transition_profile,
 lifecycle_transition_rates, lifecycle_auc_pairs, lifecycle_auc_summary) = run_lifecycle_check(
    RAW_LIFECYCLE, [("train", train), ("oos", oos), ("jan", test), ("feb", oot)],
    MODEL_LIFECYCLE, TARGET,
)
