from pathlib import Path
import json
import pandas as pd

excel_path = Path("описания.xlsx")
json_path = Path("новый_справочник.json")

# Первый лист, заголовки в первой строке.
df = pd.read_excel(
    excel_path, sheet_name=0, dtype=str, keep_default_na=False
)
df.columns = df.columns.astype(str).str.strip().str.lstrip("\ufeff")

# Названия колонок из твоего исходного файла.
columns = {
    "product": "product",
    "theme": "theme",
    "category": "subclass",
    "short_description": "description",
}

missing = set(columns) - set(df.columns)
if missing:
    raise ValueError(f"В Excel отсутствуют колонки: {sorted(missing)}")

taxonomy = []
seen = set()

for excel_row, (_, row) in enumerate(df.iterrows(), start=2):
    if not any(str(value).strip() for value in row):
        continue

    item = {
        target: str(row[source]).strip()
        for source, target in columns.items()
    }

    for field, value in item.items():
        if not value:
            raise ValueError(f"Строка {excel_row}: пустое поле {field}")

    key = tuple(
        " ".join(item[field].split())
        for field in ("product", "theme", "subclass")
    )
    if key in seen:
        raise ValueError(f"Строка {excel_row}: повтор записи {key}")
    seen.add(key)

    item["keywords"] = str(row.get("keywords", "")).strip()

    claims_used = str(row.get("claims_used", "")).strip()
    if claims_used:
        item["claims_used"] = int(claims_used)

    taxonomy.append(item)

if not taxonomy:
    raise ValueError("В Excel нет заполненных записей")

config = {
    "classifier_name": "subclass_catalog",
    "classification_mode": "product_taxonomy",
    "taxonomy_scope": "product",
    "product": None,
    "text_column": "description_claim",
    "query_path": "../queries/get_data.sql",
    "taxonomy": taxonomy,
}

# Режим x разрешает только создание нового файла.
with json_path.open("x", encoding="utf-8") as file:
    json.dump(config, file, ensure_ascii=False, indent=2)

print(f"Создан файл: {json_path.resolve()}")
print(f"Записей: {len(taxonomy)}")
