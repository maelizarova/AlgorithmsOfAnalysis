tmp_table VARCHAR2(100);

all_cred_contracts_feat0 varchar2(10000 char);
all_cred_contracts_feat1 varchar2(10000 char);

   -- Универсальная процедура для удаления таблицы, игнорируя ошибку, если таблица не существует
  PROCEDURE drop_table_if_exists(table_name IN VARCHAR2) IS
  BEGIN
    EXECUTE IMMEDIATE 'DROP TABLE ' || table_name;
  EXCEPTION
    WHEN OTHERS THEN
      IF SQLCODE = -942 THEN
        NULL;  -- Игнорируем ошибку "таблица не существует"
      ELSE
        RAISE;  -- Для других ошибок выбрасываем исключение
      END IF;
  END drop_table_if_exists;



BEGIN

tmp_table := in_table ||'_cred_contracts';

--вызов процедуры для удаления временных таблиц
drop_table_if_exists(out_table);
drop_table_if_exists(tmp_table);
drop_table_if_exists(concat(in_table, '_first_cred_prod'));
drop_table_if_exists(concat(in_table, '_last_cred_prod'));
drop_table_if_exists(concat(in_table, '_all_time_cred_contracts'));
drop_table_if_exists(concat(in_table, '_working_cred_contracts'));
drop_table_if_exists(concat(in_table, '_closed_cred_contracts'));


--таблица с промежуточными результатами
execute immediate
'
create table '||tmp_table||' as 
select /*+parallel(8)*/
    a.client_did,
    a.report_date,
    dm.contract_uid,
    dm.product_category_nm,
    dm.status_nm,
    dm.open_dt,
    dm.fact_close_dt,
    dm.contract_amt,
    dm.current_debt_amt,
    dm.current_principal_sum,
    dm.limit_available_amt,
    dm.effective_from_dt, 
    dm.effective_to_dt
from '||in_table||' a
inner join DM_MRK.DM_MRK_CONTRACT dm
on a.client_did = dm.client_did 
and open_dt < report_date
and report_date between effective_from_dt and effective_to_dt
and contract_type = ''C''
and product_category_nm != ''Перерасход''
'; 

---------------------ПЕРВЫЙ ОТКРЫТЫЙ ПРОДУКТ------------------------------------

execute immediate
'create table '||in_table||'_first_cred_prod as
select client_did, report_date, contract_cred_product_category_nm_by_open_dt_max, contract_cred_contract_amt_by_open_dt_max, contract_cred_report_date_diff_open_dt_max
from (
    select /* + parallel(8) */
    a.client_did,
    a.report_date,
    
    --категория первого открытого продукта
    b.product_category_nm as contract_cred_product_category_nm_by_open_dt_max,
    
    --сумма первого открытого продукта
    b.contract_amt as contract_cred_contract_amt_by_open_dt_max,
    
    --кол-во месяцев с первого открытого продукта
    round(months_between(a.report_date, b.open_dt)) as contract_cred_report_date_diff_open_dt_max,
    
    row_number() over (partition by a.client_did order by b.open_dt asc) as rn
    
    from '||in_table||' a
    join '||tmp_table||' b
    on a.client_did = b.client_did
) t
where rn = 1
order by client_did
';

------------------ПОСЛЕДНИЙ ОТКРЫТЫЙ ПРОДУКТ------------------------------------

execute immediate
'create table '||in_table||'_last_cred_prod as
select client_did, report_date, contract_cred_product_category_nm_by_open_dt_min, contract_cred_contract_amt_by_open_dt_min, contract_cred_report_date_diff_open_dt_min
from (
    select /* + parallel(8) */
    a.client_did,
    a.report_date,
    
    --категория последнего открытого продукта
    b.product_category_nm as contract_cred_product_category_nm_by_open_dt_min,
    
    --сумма последнего открытого продукта
    b.contract_amt as contract_cred_contract_amt_by_open_dt_min,
    
    --кол-во месяцев с последнего открытого продукта
    round(months_between(a.report_date, b.open_dt)) as contract_cred_report_date_diff_open_dt_min,
    
    row_number() over (partition by a.client_did order by open_dt desc) as rn
    
    from '||in_table||' a
    join '||tmp_table||' b
    on a.client_did = b.client_did
) t
where rn = 1
order by client_did
';
