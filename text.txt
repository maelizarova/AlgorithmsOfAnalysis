import numpy as np
import pandas as pd
from sklearn.metrics import roc_auc_score

N_TREES = [239, 800, 2985]
model_features = list(best_model.feature_name_)

assert best_model.booster_.current_iteration() >= max(N_TREES)

train_ids = set(train['client_did'])
oos_ids = set(oos['client_did'])
historical_ids = train_ids | oos_ids
january_ids = set(test['client_did'])

assert train_ids.isdisjoint(oos_ids)

# Проверяем периоды и наличие ключей.
for data, month in [(test, '2026-01'), (oot, '2026-02')]:
    assert data['client_did'].notna().all()
    assert (
        pd.to_datetime(data['report_date']).dt.to_period('M')
        == pd.Period(month, freq='M')
    ).all()


# 1. Одинаковое определение групп для января и февраля.
# "unseen_sep_dec" — отсутствовал в сентябре–декабре,
# а не обязательно получил новую карту.
def add_client_group(data):
    result = data.copy()
    result['client_group'] = np.select(
        [
            result['client_did'].isin(train_ids),
            result['client_did'].isin(oos_ids),
        ],
        ['seen_in_train', 'held_out_oos'],
        default='unseen_sep_dec'
    )
    return result


january_check = add_client_group(test)
february_check = add_client_group(oot)

periods = {
    '2026-01': january_check,
    '2026-02': february_check,
}

group_order = [
    'all',
    'seen_in_train',
    'held_out_oos',
    'unseen_sep_dec'
]


# 2. Состав выборок.
composition_rows = []

for month, data in periods.items():
    for group_name in group_order[1:]:
        group = data.loc[data['client_group'].eq(group_name)]

        composition_rows.append({
            'month': month,
            'client_group': group_name,
            'n_contract_rows': len(group),
            'share_rows': len(group) / len(data),
            'n_clients': group['client_did'].nunique(),
            'positives': group[TARGET].sum(),
            'target_share': group[TARGET].mean(),
        })

composition = pd.DataFrame(composition_rows)
display(composition.round(4))


# 3. AUC по числу деревьев, месяцам и группам.
metric_rows = []

for month, data in periods.items():
    for n in N_TREES:
        predictions = best_model.predict_proba(
            data[model_features],
            num_iteration=n
        )[:, 1]

        for group_name in group_order:
            mask = (
                np.ones(len(data), dtype=bool)
                if group_name == 'all'
                else data['client_group'].eq(group_name).to_numpy()
            )

            y = data.loc[mask, TARGET]
            auc = (
                roc_auc_score(y, predictions[mask])
                if y.nunique() == 2
                else np.nan
            )

            metric_rows.append({
                'month': month,
                'client_group': group_name,
                'n_trees': n,
                'n_samples': int(mask.sum()),
                'roc_auc': auc,
            })

metrics = pd.DataFrame(metric_rows)

comparison = metrics.pivot(
    index=['month', 'client_group'],
    columns='n_trees',
    values='roc_auc'
).reindex(
    pd.MultiIndex.from_product(
        [list(periods), group_order],
        names=['month', 'client_group']
    )
)

comparison['delta_2985_vs_239'] = comparison[2985] - comparison[239]
comparison['delta_2985_vs_800'] = comparison[2985] - comparison[800]

display(comparison.round(6))


# 4. Изменение AUC с января на февраль внутри каждой группы.
# Это сравнение групп, а не обязательно одних и тех же контрактов.
temporal_change = (
    comparison.loc['2026-02', N_TREES]
    - comparison.loc['2026-01', N_TREES]
)

temporal_change.columns = [
    f'delta_feb_minus_jan_{n}' for n in N_TREES
]

display(temporal_change.round(6))


# 5. Откуда взялись клиенты, отсутствующие в сентябре–декабре?
february_check['history_group'] = np.select(
    [
        february_check['client_did'].isin(historical_ids),
        february_check['client_did'].isin(january_ids),
    ],
    [
        'present_sep_dec',
        'absent_sep_dec_but_present_jan',
    ],
    default='absent_in_all_previous_samples'
)

history_summary = (
    february_check.groupby('history_group')
    .agg(
        n_contract_rows=(TARGET, 'size'),
        n_clients=('client_did', 'nunique'),
        target_share=(TARGET, 'mean')
    )
)

history_summary['share_rows'] = (
    history_summary['n_contract_rows'] / len(february_check)
)

display(history_summary.round(4))
