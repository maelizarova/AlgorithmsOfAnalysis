# Одна самостоятельная ячейка. Нужны исходные train, oos, test, oot,
# best_model и TARGET. test = январь, oot = февраль.
# Все месяцы анализируем отдельно; полную широкую таблицу не объединяем.
import numpy as np
import pandas as pd
from pandas.api.types import is_numeric_dtype
from sklearn.metrics import roc_auc_score
from IPython.display import display

model_monthly_check = best_model
features_monthly_check = list(model_monthly_check.booster_.feature_name())
parts_monthly_check = [("train", train), ("oos", oos), ("jan", test), ("feb", oot)]
months_monthly_check = pd.period_range("2025-09", "2026-02", freq="M").astype(str).tolist()
gain_monthly_check = pd.Series(
    model_monthly_check.booster_.feature_importance(importance_type="gain"),
    index=features_monthly_check,
)

# Объединяем только ключи, дату и таргет; признаки далее читаем по одному.
meta_monthly_check = pd.concat([
    frame[["contract_uid", "client_did", "report_date", TARGET]].assign(source=name)
    for name, frame in parts_monthly_check
], ignore_index=True)
meta_monthly_check["report_date"] = pd.to_datetime(meta_monthly_check["report_date"])
meta_monthly_check["month"] = meta_monthly_check["report_date"].dt.strftime("%Y-%m")
assert meta_monthly_check[["contract_uid", "report_date", TARGET]].notna().all().all()
assert meta_monthly_check[TARGET].isin([0, 1]).all(), "Нужен бинарный таргет"
assert set(meta_monthly_check["month"]) == set(months_monthly_check), "Ожидались сентябрь–февраль"
assert not meta_monthly_check.duplicated(["contract_uid", "month"]).any(), "Повтор контракта внутри месяца"
assert meta_monthly_check.loc[meta_monthly_check.source.isin(["train", "oos"]), "month"].lt("2026-01").all()
assert meta_monthly_check.loc[meta_monthly_check.source.eq("jan"), "month"].eq("2026-01").all()
assert meta_monthly_check.loc[meta_monthly_check.source.eq("feb"), "month"].eq("2026-02").all()
indices_monthly_check = {
    month: meta_monthly_check.index[meta_monthly_check["month"].eq(month)].to_numpy()
    for month in months_monthly_check
}
y_monthly_check = meta_monthly_check[TARGET].to_numpy(dtype=int)

def auc_monthly_check(y, score):
    return roc_auc_score(y, score) if np.unique(y).size == 2 else np.nan

# 1. Объём и таргет: все строки каждого месяца, включая train и OOS.
monthly_target_summary = meta_monthly_check.groupby("month").agg(
    n_contracts=("contract_uid", "size"),
    n_clients=("client_did", "nunique"),
    target_pct=(TARGET, "mean"),
)
monthly_target_summary["target_pct"] *= 100
monthly_target_summary["delta_target_pp"] = monthly_target_summary["target_pct"].diff()
print("1. Состав и доля таргета по месяцам:")
display(monthly_target_summary.round(3))

# AUC модели: для сентября–декабря только исходный OOS.
# Модель не обучаем. Предсказываем на исходных фреймах, сохраняя типы категорий.
model_rows_monthly_check = []
for name, frame in parts_monthly_check:
    if name == "train":
        continue
    score = model_monthly_check.predict_proba(frame[features_monthly_check])[:, 1]
    positions = np.flatnonzero(meta_monthly_check.source.eq(name).to_numpy())
    for month in months_monthly_check:
        mask = meta_monthly_check.iloc[positions]["month"].eq(month).to_numpy()
        if mask.any():
            model_rows_monthly_check.append({
                "month": month, "sample": {"oos": "OOS", "jan": "validation", "feb": "OOT"}[name],
                "n_rows": int(mask.sum()),
                "auc": auc_monthly_check(y_monthly_check[positions][mask], score[mask]),
            })
monthly_model_auc = pd.DataFrame(model_rows_monthly_check).set_index("month").sort_index()
print("AUC одной исходной модели: OOS → январская валидация → февраль:")
display(monthly_model_auc.round(6))

# 2. Каждый переход — на общих контрактах ДВУХ соседних месяцев.
# Пересечение всех шести месяцев не требуется.
pair_indices_monthly_check = []
transition_rows_monthly_check, pair_rows_monthly_check = [], []
for before, after in zip(months_monthly_check[:-1], months_monthly_check[1:]):
    lookups = []
    for month in [before, after]:
        idx = indices_monthly_check[month]
        lookups.append(pd.Series(idx, index=meta_monthly_check.iloc[idx]["contract_uid"]))
    common = lookups[0].index.intersection(lookups[1].index)
    assert len(common), f"Нет общих контрактов: {before} → {after}"
    ia, ib = [lookup.loc[common].to_numpy() for lookup in lookups]
    pair = f"{before} → {after}"
    pair_indices_monthly_check.append((pair, ia, ib))
    ya, yb = y_monthly_check[ia], y_monthly_check[ib]
    n01 = int(((ya == 0) & (yb == 1)).sum())
    n10 = int(((ya == 1) & (yb == 0)).sum())
    pair_rows_monthly_check.append({
        "pair": pair, "n_common": len(common),
        "common_before_pct": 100 * len(common) / len(lookups[0]),
        "common_after_pct": 100 * len(common) / len(lookups[1]),
        "target_before_pct": 100 * ya.mean(), "target_after_pct": 100 * yb.mean(),
        "changed_pct": 100 * (ya != yb).mean(),
        "n_0_to_1": n01, "n_1_to_0": n10,
        "new_negatives_pct": 100 * n10 / (yb == 0).sum() if (yb == 0).any() else np.nan,
    })
    for a, b in [(0, 0), (0, 1), (1, 0), (1, 1)]:
        count = int(((ya == a) & (yb == b)).sum())
        class_n = int((ya == a).sum())
        transition_rows_monthly_check.append({
            "pair": pair, "transition": f"{a} → {b}", "n_contracts": count,
            "share_all_pct": 100 * count / len(common),
            "share_within_before_class_pct": 100 * count / class_n if class_n else np.nan,
        })
monthly_transition_summary = pd.DataFrame(pair_rows_monthly_check).set_index("pair")
monthly_target_transitions = pd.DataFrame(transition_rows_monthly_check)
print("2. Переходы на общих контрактах. new_negatives_pct: доля бывших единиц среди нулей следующего месяца:")
display(monthly_transition_summary[[
    "n_common", "target_before_pct", "target_after_pct", "changed_pct",
    "n_0_to_1", "n_1_to_0", "new_negatives_pct",
]].round(3))
# Подробности четырёх переходов: display(monthly_target_transitions.round(3))

# 3. Все признаки и таргет СВОЕГО месяца.
# AUC отдельно на полном месяце и на одинаковых парах соседних месяцев.
# Это описание связи в данных; выбранные признаки не считаются независимым тестом модели.
auc_rows_monthly_check, paired_rows_monthly_check, bin_rows_monthly_check = [], [], []
categories_monthly_check = set(globals().get("categorical_features", []))
for feature in features_monthly_check:
    numeric = feature not in categories_monthly_check and all(
        is_numeric_dtype(frame[feature].dtype) for _, frame in parts_monthly_check
    )
    values = pd.concat([frame[feature] for _, frame in parts_monthly_check], ignore_index=True)
    gain_pct = 100 * gain_monthly_check[feature] / max(gain_monthly_check.sum(), 1e-12)
    if numeric:
        v = values.to_numpy(dtype=float, na_value=np.nan)
        # ОДНИ границы по сентябрю для всех шести месяцев.
        base = v[indices_monthly_check["2025-09"]]
        base = base[np.isfinite(base)]
        cuts = np.unique(np.quantile(base, [0.2, 0.4, 0.6, 0.8])) if len(base) else []
        edges = np.r_[-np.inf, cuts, np.inf]
        for pair, ia, ib in pair_indices_monthly_check:
            valid = np.isfinite(v[ia]) & np.isfinite(v[ib])
            aa = auc_monthly_check(y_monthly_check[ia][valid], v[ia][valid])
            ab = auc_monthly_check(y_monthly_check[ib][valid], v[ib][valid])
            paired_rows_monthly_check.append({
                "feature": feature, "pair": pair, "n_pairs": int(valid.sum()),
                "auc_before": aa, "auc_after": ab,
                "delta_strength": abs(ab - .5) - abs(aa - .5),
                "direction_flip": (aa-.5)*(ab-.5) < 0 if np.isfinite(aa) and np.isfinite(ab) else pd.NA,
            })
    else:
        values = values.astype("string").fillna("<MISSING>")

    for month, idx in indices_monthly_check.items():
        if numeric:
            finite = np.isfinite(v[idx])
            auc_rows_monthly_check.append({
                "feature": feature, "month": month, "gain_pct": gain_pct,
                "n_valid": int(finite.sum()), "missing_or_inf_pct": 100 * (~finite).mean(),
                "auc": auc_monthly_check(y_monthly_check[idx][finite], v[idx][finite]),
            })
            bucket = pd.cut(pd.Series(v[idx]).where(finite), edges, precision=6)
            bucket = bucket.cat.add_categories(["MISSING/INF"]).fillna("MISSING/INF")
        else:
            bucket = values.iloc[idx].reset_index(drop=True)
        profile = pd.DataFrame({"bucket": bucket, "y": y_monthly_check[idx]}).groupby(
            "bucket", observed=True, sort=False, dropna=False
        ).agg(n_rows=("y", "size"), target_pct=("y", "mean")).reset_index()
        profile["target_pct"] *= 100
        profile["share_rows_pct"] = 100 * profile["n_rows"] / len(idx)
        profile["feature"], profile["month"] = feature, month
        bin_rows_monthly_check.append(profile)

monthly_feature_auc = pd.DataFrame(auc_rows_monthly_check)
paired_feature_auc = pd.DataFrame(paired_rows_monthly_check)
monthly_feature_bins = pd.concat(bin_rows_monthly_check, ignore_index=True)
monthly_feature_auc_wide = pd.DataFrame()
if not monthly_feature_auc.empty:
    monthly_feature_auc_wide = monthly_feature_auc.pivot(index="feature", columns="month", values="auc")
    order = gain_monthly_check.sort_values(ascending=False).index.intersection(monthly_feature_auc_wide.index, sort=False)
    monthly_feature_auc_wide = monthly_feature_auc_wide.reindex(order)
print("3. AUC числовых признаков с таргетом своего месяца; все контракты месяца:")
display(monthly_feature_auc_wide.round(6))
print("Изменение силы связи на общих контрактах соседних месяцев (отрицательное = ослабление):")
if not paired_feature_auc.empty:
    display(paired_feature_auc.pivot(index="feature", columns="pair", values="delta_strength")
            .reindex(monthly_feature_auc_wide.index).round(6))

# Компактный вывод фиксированных интервалов для ключевых признаков.
# Полные данные по всем 40 признакам, включая категории: monthly_feature_bins.
print("Доля таргета по фиксированным интервалам; исходные численности — в monthly_feature_bins:")
show_monthly_check = list(dict.fromkeys(

for feature in show_monthly_check:

    if not table.empty:
        print(feature)
        order = table["bucket"].drop_duplicates().tolist()
        display(table.pivot(index="bucket", columns="month", values="target_pct").reindex(order).round(3))

# 4. Опционально: исходная непрерывная доля будущего среднего ОД / текущего лимита.
# Нужна таблица с contract_uid, report_date и колонкой доли; 0.25 означает 25%.
def check_continuous_target_monthly(raw_data, value_col, threshold=0.25, band=0.05):
    assert 0 < band < threshold
    source = raw_data[["contract_uid", "report_date", value_col]].copy()
    source["report_date"] = pd.to_datetime(source["report_date"])
    assert not source.duplicated(["contract_uid", "report_date"]).any(), "Повтор контракта × даты"
    data = meta_monthly_check[["contract_uid", "report_date", "month"]].merge(
        source, on=["contract_uid", "report_date"], how="left", validate="one_to_one", indicator=True,
    )
    data["ratio"] = pd.to_numeric(data[value_col], errors="raise").astype(float).replace([np.inf, -np.inf], np.nan)
    edges = [-np.inf, 0, threshold-band, threshold, threshold+band, 2*threshold, 4*threshold, np.inf]
    rows, distributions = [], []
    for month, frame in data.groupby("month", sort=True):
        finite = frame["ratio"].dropna()
        row = {
            "month": month, "n_rows": len(frame), "matched_keys_pct": 100 * frame["_merge"].eq("both").mean(),
            "missing_or_inf_pct": 100 * frame["ratio"].isna().mean(), "mean": finite.mean(),
            "near_threshold_pct": 100 * finite.between(threshold-band, threshold+band).mean(),
            "above_threshold_pct": 100 * finite.gt(threshold).mean(),
        }
        row.update({f"q{q:02d}": finite.quantile(q/100) for q in [1, 5, 25, 50, 75, 95, 99]})
        rows.append(row)
        bucket = pd.cut(frame["ratio"], edges, precision=6).cat.add_categories(["MISSING/INF"]).fillna("MISSING/INF")
        counts = bucket.value_counts(sort=False)
        distributions.append(pd.DataFrame({"month": month, "bucket": counts.index,
                                           "n_rows": counts.to_numpy(), "share_pct": 100*counts.to_numpy()/len(frame)}))
    summary, distribution = pd.DataFrame(rows).set_index("month"), pd.concat(distributions, ignore_index=True)
    print("Непрерывный таргет: все строки месяца; near/above — среди конечных заполненных значений:")
    display(summary.round(4))
    display(distribution.pivot(index="bucket", columns="month", values="share_pct")
            .reindex(distribution["bucket"].drop_duplicates()).round(3))
    return summary, distribution
