DECLARE
    PROCEDURE drop_table_if_exists(table_name IN VARCHAR2) IS
    BEGIN
        EXECUTE IMMEDIATE 'DROP TABLE ' || table_name;
    EXCEPTION
        WHEN OTHERS THEN
            IF SQLCODE = -942 THEN
                NULL;
            ELSE
                RAISE;
            END IF;
    END drop_table_if_exists;

BEGIN
    drop_table_if_exists('ema_sample_CLF_PTB_CLIP_0926_V4_DEBT');

    execute immediate
    'create table ema_sample_CLF_PTB_CLIP_0926_V4_DEBT as
    select a.*,
           case when c.CURRENT_DEBT_AMT > 0 then 1 else 0 end CURRENT_DEBT_AMT_FLG
    from (
        select *
        from (
            select q.client_did,
                   q.contract_id as contract_src_id,
                   q.cur_cc_mpay*20 as old_limit,
                   r.limit as new_limit,
                   r.pdn,
                   r.decision_code,
                   r.decision_date,
                   trunc(sysdate) - 1 as report_date,
                   row_number() over (
                       partition by q.client_did
                       order by r.decision_date desc, dbms_random.value()
                   ) rn
            from integro.int_cvmrtm_request@cvm q
            join integro.int_rtmcvm_result@cvm r on r.request_id = q.request_id
            where 1=1
                and q.in_product_type = ''CLIP''
                and q.int_create_date >= trunc(sysdate) - 1
                and q.int_create_date < trunc(sysdate)
                and nvl(q.int_status, ''A'') <> ''ERROR''
                and q.campaign_code <> ''FICO_TEST''
        )
        where rn = 1
            and decision_code = ''ACCEPT''
            and pdn <= 0.8
            and new_limit > old_limit
    ) a
    left join DM_MRK.DM_MRK_CONTRACT c
        on a.client_did = c.client_did
        and a.contract_src_id = c.CONTRACT_SRC_ID
        and trunc(a.report_date) between c.EFFECTIVE_FROM_DT and c.EFFECTIVE_TO_DT
        and c.contract_type = ''C''';
END;
