# Одна ячейка после предыдущих проверок. Нужны sample_check, test, oot,
# best_model, TARGET. Модели не переобучаются, исходные фреймы не меняются.
# target = будущая средняя задолженность; current_limit = лимит для target_25.
import numpy as np
import pandas as pd
from sklearn.metrics import roc_auc_score
from IPython.display import display

FLIP_THRESHOLD = 0.25
FLIP_BAND = 0.05
FLIP_DEBT_COL = "target"
FLIP_LIMIT_COL = "current_limit"
FLIP_KEYS = ["contract_uid", "report_date"]

# Копируем только ключи и две величины, а не всю таблицу с признаками.
raw_flip_check = sample_check[FLIP_KEYS + [FLIP_DEBT_COL, FLIP_LIMIT_COL]].copy()
raw_flip_check["report_date"] = pd.to_datetime(raw_flip_check["report_date"])
raw_flip_check = raw_flip_check.loc[
    raw_flip_check["report_date"].ge("2026-01-01")
    & raw_flip_check["report_date"].lt("2026-03-01")
].rename(columns={FLIP_DEBT_COL: "future_debt", FLIP_LIMIT_COL: "limit"})
for col in ["future_debt", "limit"]:
    raw_flip_check[col] = raw_flip_check[col].astype("float64")
assert not raw_flip_check.duplicated(FLIP_KEYS).any(), "Повтор контракт × дата в sample_check"

sides_flip_check = []
for month, frame in [("2026-01", test), ("2026-02", oot)]:
    base = frame[FLIP_KEYS + [TARGET]].copy()
    base["report_date"] = pd.to_datetime(base["report_date"])
    assert base["report_date"].dt.strftime("%Y-%m").eq(month).all()
    assert base["contract_uid"].notna().all() and base["contract_uid"].is_unique
    assert base[TARGET].isin([0, 1]).all()
    side = base.merge(raw_flip_check, on=FLIP_KEYS, how="left", validate="one_to_one", indicator=True)
    assert side["_merge"].eq("both").all(), f"Не все ключи найдены в sample_check: {month}"
    sides_flip_check.append(side.drop(columns="_merge").rename(columns={TARGET: "y"}).set_index("contract_uid"))

# Одинаковые контракты, исходные бинарные метки сохраняем.
flip_pairs = sides_flip_check[0].join(sides_flip_check[1], how="inner", lsuffix="_jan", rsuffix="_feb")
assert len(flip_pairs), "Нет общих контрактов"
flip_pairs["transition"] = flip_pairs.y_jan.astype(int).astype(str) + " -> " + flip_pairs.y_feb.astype(int).astype(str)
for month in ["jan", "feb"]:
    valid = (
        np.isfinite(flip_pairs["future_debt_" + month])
        & np.isfinite(flip_pairs["limit_" + month])
        & flip_pairs["limit_" + month].gt(0)
    )
    flip_pairs["ratio_" + month] = (
        flip_pairs["future_debt_" + month] / flip_pairs["limit_" + month].where(valid)
    )
flip_pairs["valid_ratios"] = np.isfinite(flip_pairs.ratio_jan) & np.isfinite(flip_pairs.ratio_feb)

features_flip_check = list(best_model.booster_.feature_name())
for month, frame in [("jan", test), ("feb", oot)]:
    positions = pd.Index(frame["contract_uid"]).get_indexer(flip_pairs.index)
    assert (positions >= 0).all()
    flip_pairs["score_" + month] = best_model.predict_proba(frame[features_flip_check].iloc[positions])[:, 1]

flip_pairs["ratio_delta"] = flip_pairs.ratio_feb - flip_pairs.ratio_jan
flip_pairs["debt_delta"] = flip_pairs.future_debt_feb - flip_pairs.future_debt_jan
flip_pairs["limit_delta_pct"] = 100 * (flip_pairs.limit_feb / flip_pairs.limit_jan.where(flip_pairs.limit_jan > 0) - 1)
for month in ["jan", "feb"]:
    flip_pairs["top20_" + month] = flip_pairs["score_" + month].ge(flip_pairs["score_" + month].quantile(.8))

transition_order_flip_check = ["0 -> 0", "0 -> 1", "1 -> 0", "1 -> 1"]
flip_summary = flip_pairs.groupby("transition").agg(
    n_contracts=("y_jan", "size"),
    ratio_jan_median=("ratio_jan", "median"), ratio_feb_median=("ratio_feb", "median"),
    ratio_delta_median=("ratio_delta", "median"),
    score_jan_mean=("score_jan", "mean"), score_feb_mean=("score_feb", "mean"),
    top20_jan_pct=("top20_jan", "mean"), top20_feb_pct=("top20_feb", "mean"),
).reindex(transition_order_flip_check)
flip_summary["n_contracts"] = flip_summary["n_contracts"].fillna(0).astype(int)
flip_summary[["top20_jan_pct", "top20_feb_pct"]] *= 100
print(f"Общих контрактов: {len(flip_pairs):,}. 1. Все переходы для сравнения:")
display(flip_summary.round(4))

# Проверяем соответствие отношения исходной бинарной метке.
# Редкие расхождения на границе не меняем автоматически.
rule_rows_flip_check = []
for month in ["jan", "feb"]:
    valid = np.isfinite(flip_pairs["ratio_" + month])
    mismatch = flip_pairs.loc[valid, "ratio_" + month].gt(FLIP_THRESHOLD).ne(flip_pairs.loc[valid, "y_" + month])
    rule_rows_flip_check.append({"month": month, "n_valid": int(valid.sum()),
                                 "n_rule_mismatches": int(mismatch.sum()), "mismatch_pct": 100*mismatch.mean()})
flip_rule_check = pd.DataFrame(rule_rows_flip_check)
display(flip_rule_check.round(5))

# 2. Подробно исследуем реальные 1 -> 0.
flips_10_check = flip_pairs.loc[flip_pairs.transition.eq("1 -> 0")].copy()
valid_10_check = flips_10_check.loc[flips_10_check.valid_ratios].copy()
print(f"2. Переходов 1 -> 0: {len(flips_10_check):,}; с корректными отношениями: {len(valid_10_check):,}")
flip_value_quantiles = valid_10_check[[
    "ratio_jan", "ratio_feb", "ratio_delta", "future_debt_jan", "future_debt_feb", "debt_delta",
    "limit_jan", "limit_feb", "limit_delta_pct",
]].quantile([.1, .5, .9]).T.rename(columns={.1: "q10", .5: "median", .9: "q90"})
display(flip_value_quantiles.round(4))

near_jan_flip_check = valid_10_check.ratio_jan.between(FLIP_THRESHOLD-FLIP_BAND, FLIP_THRESHOLD+FLIP_BAND)
near_feb_flip_check = valid_10_check.ratio_feb.between(FLIP_THRESHOLD-FLIP_BAND, FLIP_THRESHOLD+FLIP_BAND)
conditions_flip_check = {
    "Оба значения в диапазоне 0.20–0.30": near_jan_flip_check & near_feb_flip_check,
    "Хотя бы одно значение в диапазоне 0.20–0.30": near_jan_flip_check | near_feb_flip_check,
    "Январское отношение > 0.50, февральское <= 0.25": valid_10_check.ratio_jan.gt(.5) & valid_10_check.ratio_feb.le(FLIP_THRESHOLD),
    "Будущая средняя задолженность уменьшилась": valid_10_check.debt_delta.lt(0),
    "Лимит на дату скоринга вырос": valid_10_check.limit_feb.gt(valid_10_check.limit_jan),
}
flip_ratio_details = pd.DataFrame([
    {"condition": name, "n_contracts": int(mask.sum()), "share_valid_flips_pct": 100*mask.mean()}
    for name, mask in conditions_flip_check.items()
]).set_index("condition")
print("Условия могут пересекаться; проценты считаются внутри 1 -> 0 с корректными отношениями:")
display(flip_ratio_details.round(3))

# 3. Арифметика порога: меняем по одному компоненту отношения.
# Это НЕ причинная оценка эффекта выдачи лимита или изменения поведения.
# Будущие долги относятся к разным окнам дней 151–180 после дат скоринга.
mechanism_input_flip_check = valid_10_check.loc[
    valid_10_check.ratio_jan.gt(FLIP_THRESHOLD) & valid_10_check.ratio_feb.le(FLIP_THRESHOLD)
].copy()
debt_only_flip_check = (mechanism_input_flip_check.future_debt_feb / mechanism_input_flip_check.limit_jan).le(FLIP_THRESHOLD)
limit_only_flip_check = (mechanism_input_flip_check.future_debt_jan / mechanism_input_flip_check.limit_feb).le(FLIP_THRESHOLD)
mechanism_masks_flip_check = {
    "Только изменения будущего долга отдельно достаточно": debt_only_flip_check & ~limit_only_flip_check,
    "Только изменения лимита отдельно достаточно": ~debt_only_flip_check & limit_only_flip_check,
    "Каждого изменения отдельно достаточно": debt_only_flip_check & limit_only_flip_check,
    "Порог пересекается только при совместном изменении": ~debt_only_flip_check & ~limit_only_flip_check,
}
flip_mechanisms = pd.DataFrame([
    {"mechanism": name, "n_contracts": int(mask.sum()), "share_analyzed_flips_pct": 100*mask.mean()}
    for name, mask in mechanism_masks_flip_check.items()
]).set_index("mechanism")
print(f"3. Разложение пересечения порога; учтено {len(mechanism_input_flip_check):,} контрактов:")
display(flip_mechanisms.round(3))

# 4. Как новые нули ранжируются относительно ОДНОЙ группы февральских единиц.
# Внутри группы 1 -> 0 AUC не определён: все февральские метки равны нулю.
# Поэтому сравниваем с общим пулом февральских положительных исходов.
positive_flip_check = flip_pairs.y_feb.eq(1)
n_negative_flip_check = int(flip_pairs.y_feb.eq(0).sum())
auc_rows_flip_check = []
def safe_auc_flip_check(y, score):
    return roc_auc_score(y, score) if y.nunique() == 2 else np.nan

for transition in ["0 -> 0", "1 -> 0"]:
    negative = flip_pairs.transition.eq(transition)
    mask = positive_flip_check | negative
    auc_rows_flip_check.append({
        "negative_group": transition, "n_negative": int(negative.sum()),
        "share_feb_negatives_pct": 100*negative.sum()/n_negative_flip_check if n_negative_flip_check else np.nan,
        "auc_jan_score": safe_auc_flip_check(flip_pairs.loc[mask, "y_feb"], flip_pairs.loc[mask, "score_jan"]),
        "auc_feb_score": safe_auc_flip_check(flip_pairs.loc[mask, "y_feb"], flip_pairs.loc[mask, "score_feb"]),
    })
flip_negative_auc = pd.DataFrame(auc_rows_flip_check).set_index("negative_group")
print("4. В обоих сравнениях используются все февральские единицы общих контрактов:")
display(flip_negative_auc.round(6))

auc_common_feb_flip_check = safe_auc_flip_check(flip_pairs.y_feb, flip_pairs.score_feb)
if positive_flip_check.any() and n_negative_flip_check:
    weighted_auc_flip_check = (
        flip_negative_auc["auc_feb_score"] * flip_negative_auc["n_negative"]
    ).sum() / n_negative_flip_check
    assert np.isclose(weighted_auc_flip_check, auc_common_feb_flip_check), "Не сошлось разложение AUC"
    print(f"Февральский AUC на всех общих контрактах: {auc_common_feb_flip_check:.6f}")
    print(f"Взвешенная сумма двух сравнений:           {weighted_auc_flip_check:.6f}")
    print("Это разложение февральской метрики, а не оценка причины разницы между январём и февралём.")

# Короткий текст для добавления в общие выводы после запуска.
print("\nЧИСЛА ДЛЯ ОБЩЕГО ВЫВОДА:")
print(f"Из {len(flip_pairs):,} общих контрактов {len(flips_10_check):,} перешли 1 -> 0.")
if n_negative_flip_check:
    print(f"Их доля среди февральских нулей: {100*len(flips_10_check)/n_negative_flip_check:.2f}%.")
if len(valid_10_check):
    print(f"Медиана отношения у них: {valid_10_check.ratio_jan.median():.4f} -> {valid_10_check.ratio_feb.median():.4f}.")
    print(f"Оба значения в диапазоне 0.20–0.30: {100*(near_jan_flip_check & near_feb_flip_check).mean():.2f}%.")
    print(f"Из отношения > 0.50 в <= 0.25 перешли {100*conditions_flip_check['Январское отношение > 0.50, февральское <= 0.25'].mean():.2f}%.")
if len(flips_10_check):
    print(f"В верхние 20% январского ранжирования попадали {100*flips_10_check.top20_jan.mean():.2f}% этой группы;")
    print(f"в верхние 20% февральского ранжирования — {100*flips_10_check.top20_feb.mean():.2f}%.")
print("Группы верхних 20% определены на общих контрактах; при совпадающих скорах могут быть чуть больше 20%.")
