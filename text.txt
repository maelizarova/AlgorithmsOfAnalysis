import plotly.graph_objects as go

_REPORT_SPLITS = {'train': 'train', 'test': 'test', 'oot': 'oot'}
_REPORT_ORDER = {'train': 0, 'test': 1, 'oot': 2}

def _report_table(data, columns, names, percent_columns=(), integer_columns=()):
    table = data.loc[:, columns].copy()
    if 'segment' in table:
        table = table.sort_values('segment', key=lambda s: s.map(_REPORT_ORDER))
        table['segment'] = table['segment'].replace(_REPORT_SPLITS)
    for col in columns:
        if col in percent_columns:
            table[col] = table[col].map(lambda x: f'{100*x:.2f}' if pd.notna(x) else '—')
        elif col in integer_columns:
            table[col] = table[col].map(lambda x: f'{x:,.0f}'.replace(',', ' ') if pd.notna(x) else '—')
        elif pd.api.types.is_numeric_dtype(table[col]):
            table[col] = table[col].map(lambda x: f'{x:,.1f}'.replace(',', ' ') if pd.notna(x) else '—')
    return table.rename(columns=names).reset_index(drop=True)


def _report_plot(source, x_title, y_title, percent=False):
    fig = go.Figure(source)  # исходный график остаётся без изменений
    fig.update_layout(
        title=None, width=None, height=360, autosize=True,
        xaxis_title=x_title, yaxis_title=y_title,
        legend=dict(title_text='', orientation='h', x=0, y=-0.23,
                    xanchor='left', yanchor='top'),
        margin=dict(l=65, r=25, t=25, b=100))
    if percent:
        fig.update_yaxes(tickformat='.1%')
    if x_title == 'Месяц':
        fig.update_xaxes(tickformat='%m.%Y')
    for trace in fig.data:
        trace.name = _REPORT_SPLITS.get(trace.name, trace.name)
        trace.hovertemplate = (
            x_title + ': %{x}<br>' + y_title +
            (': %{y:.2%}' if percent else ': %{y:,.1f}') + '<extra>%{fullData.name}</extra>')
    for annotation in fig.layout.annotations or ():
        if annotation.text and annotation.text.startswith('Цель'):
            annotation.text = annotation.text.replace('Цель', 'Уровень')
    return fig
