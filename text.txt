# Используем только признаки текущей модели и служебные колонки.
features_once = list(best_model.feature_name_)
cols_once = list(dict.fromkeys(
    PRIMARY_KEY + ['year_month', TARGET] + features_once
))

dev_once = pd.concat(
    [train[cols_once], oos[cols_once]],
    ignore_index=True,
)

assert dev_once['contract_uid'].notna().all()
assert (
    (dev_once['report_date'] >= '2025-09-01')
    & (dev_once['report_date'] < '2026-01-01')
).all(), 'В train/oos есть строки вне сентября–декабря'

# Выбор строки контракта не зависит от таргета.
dev_once = (
    dev_once.sample(frac=1, random_state=SEED)
    .drop_duplicates('contract_uid', keep='first')
    .reset_index(drop=True)
)

# OOS: около 20%, с сохранением пропорций месяцев и классов.
oos_once = dev_once.groupby(
    [dev_once['report_date'].dt.to_period('M'), TARGET],
    observed=True,
    sort=True,
).sample(frac=0.20, random_state=SEED)

train_once = dev_once.drop(index=oos_once.index)

assert not train_once['contract_uid'].isin(
    oos_once['contract_uid']
).any()

print(f'Строк было: {len(train) + len(oos):,}')
print(f'Уникальных контрактов: {len(dev_once):,}')
print(f'Новый train: {len(train_once):,}')
print(f'Новый OOS: {len(oos_once):,}')

del dev_once
gc.collect()
