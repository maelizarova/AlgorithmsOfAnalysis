# 2.3. Hyperopt: прежнее пространство поиска,
# но early stopping и loss считаются по jan_valid_group.
def objective_group(params):
    model = lgb.LGBMClassifier(**make_model_params(params))

    model.fit(
        train_group[selected_group_features],
        train_group[TARGET],
        eval_set=[(
            jan_valid_group[selected_group_features],
            jan_valid_group[TARGET]
        )],
        eval_metric='auc',
        callbacks=[
            lgb.early_stopping(
                EARLY_STOPPING_ROUNDS,
                first_metric_only=True,
                verbose=False
            )
        ]
    )

    n_trees = int(model.best_iteration_)
    pred = model.predict_proba(
        jan_valid_group[selected_group_features],
        num_iteration=n_trees
    )[:, 1]

    return {
        'loss': -float(roc_auc_score(jan_valid_group[TARGET], pred)),
        'status': STATUS_OK,
        'best_iteration': n_trees
    }


trials_group = Trials()

best_group = fmin(
    fn=objective_group,
    space=space,
    algo=tpe.suggest,
    max_evals=100,
    trials=trials_group,
    rstate=np.random.default_rng(SEED)
)

winner_group = trials_group.best_trial['result']

params_group = make_model_params(
    space_eval(space, best_group),
    n_estimators=int(winner_group['best_iteration'])
)
